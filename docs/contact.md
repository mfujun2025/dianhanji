## 常见问题

### Q1：多久能收到回复？

我们会尽快处理收到的留言。若较为紧急，建议同时发送邮件至 **mfujun@agent.qq.com**，并在标题写明主题（如"选型咨询-碳钢6-12mm"）。

### Q2：留言需要提供手机号吗？

不强制。但留下可靠的联系方式能让我们更高效地回复。若不方便留手机号，留邮箱或微信号也可以。

### Q3：能直接报价吗？

本站聚焦选型与技术咨询。由于设备配置差异大（机型、配件、气瓶、耗材），脱离具体工况的报价没有参考价值。请先在留言中写明工况，我们再给出匹配建议。

### Q4：表单提交失败怎么办？

先检查必填项（称呼、联系方式、留言内容）是否填写完整。若提示"接收端未配置"或网络异常，请点击**「复制留言内容」**按钮，把内容直接发到 **mfujun@agent.qq.com**，不会影响您的咨询。

<script setup>
import { onMounted } from 'vue'

onMounted(() => {
  // 接收端配置：留空则进入「演示模式」（只做本地校验，可复制内容自行发送）
  // 如需自动推送到飞书/企业微信群机器人，把 webhook 地址填入下方即可。
  var CFG = {
    webhook: '',
    keyword: '留言',
    site: '电焊机.cn',
    mailto: 'mfujun@agent.qq.com',
  }

  var F = {
    name: document.getElementById('f-name'),
    contact: document.getElementById('f-contact'),
    material: document.getElementById('f-material'),
    thickness: document.getElementById('f-thickness'),
    power: document.getElementById('f-power'),
    type: document.getElementById('f-type'),
    message: document.getElementById('f-message'),
    hp: document.getElementById('dhj-hp'),
  }
  var statusEl = document.getElementById('dhj-status')
  var submitBtn = document.getElementById('dhj-submit')
  var copyBtn = document.getElementById('dhj-copy')
  var form = document.getElementById('dhj-form')

  var FIELDS = [
    ['称呼', 'name'],
    ['联系方式', 'contact'],
    ['母材材质', 'material'],
    ['主要板厚', 'thickness'],
    ['供电条件', 'power'],
    ['咨询类型', 'type'],
    ['留言内容', 'message'],
  ]

  function val(k) {
    var el = F[k]
    return el && el.value ? String(el.value).trim() : ''
  }

  function esc(s) {
    return String(s == null ? '' : s).replace(/[&<>"]/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]
    })
  }

  function nowStr() {
    var d = new Date()
    function p(n) { return n < 10 ? '0' + n : '' + n }
    return d.getFullYear() + '-' + p(d.getMonth() + 1) + '-' + p(d.getDate()) +
      ' ' + p(d.getHours()) + ':' + p(d.getMinutes())
  }

  function setStatus(msg, kind) {
    statusEl.className = 'dhj-status ' + (kind || '')
    statusEl.innerHTML = msg
  }

  function clearErrors() {
    var errs = document.querySelectorAll('.dhj-err')
    for (var i = 0; i < errs.length; i++) errs[i].textContent = ''
  }

  function collect() {
    var d = {}
    for (var i = 0; i < FIELDS.length; i++) d[FIELDS[i][1]] = val(FIELDS[i][1])
    return d
  }

  var PHONE_RE = /^1[3-9]\d{9}$/
  var MAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/

  function validate(d) {
    clearErrors()
    var ok = true
    function fail(key, msg) {
      var el = document.querySelector('.dhj-err[data-for="' + key + '"]')
      if (el) el.textContent = msg
      ok = false
    }
    if (!d.name) fail('name', '请填写称呼')
    else if (d.name.length > 30) fail('name', '称呼请控制在 30 字以内')

    if (!d.contact) fail('contact', '请填写联系方式')
    else if (d.contact.length > 60) fail('contact', '联系方式过长')

    if (!d.message) fail('message', '请填写留言内容')
    else if (d.message.length < 5) fail('message', '留言内容请不少于 5 个字')
    else if (d.message.length > 1000) fail('message', '留言内容请控制在 1000 字以内')

    return ok
  }

  function plainText(d) {
    var lines = []
    for (var i = 0; i < FIELDS.length; i++) {
      var label = FIELDS[i][0], key = FIELDS[i][1]
      if (d[key]) lines.push(label + '：' + d[key])
    }
    lines.push('来源：电焊机.cn 联系页面')
    lines.push('时间：' + nowStr())
    return lines.join('\n')
  }

  function buildCard(d) {
    var body = []
    for (var i = 0; i < FIELDS.length; i++) {
      var label = FIELDS[i][0], key = FIELDS[i][1]
      if (d[key]) body.push('**' + label + '**：' + esc(d[key]))
    }
    return {
      msg_type: 'interactive',
      card: {
        config: { wide_screen_mode: true },
        header: {
          template: 'red',
          title: { tag: 'plain_text', content: '🔔 新' + CFG.keyword + ' · ' + CFG.site },
        },
        elements: [
          { tag: 'div', text: { tag: 'lark_md', content: body.join('\n') || '（未填写）' } },
          { tag: 'hr' },
          {
            tag: 'note',
            elements: [{
              tag: 'plain_text',
              content: '来源：联系页面 · ' + nowStr() + ' · 类型：' + CFG.keyword,
            }],
          },
        ],
      },
    }
  }

  function demoMode(d) {
    setStatus(
      '留言已生成（当前为演示模式，接收端未配置）。<br>' +
      '请点击「复制留言内容」，发送至 <strong>' + esc(CFG.mailto) + '</strong>。',
      'info'
    )
  }

  function showOk(d) {
    setStatus('提交成功，感谢您的留言！我们会尽快联系您。', 'ok')
    form.reset()
  }

  function showErr(msg) {
    setStatus('提交失败：' + esc(msg) + '<br>可点击「复制留言内容」，直接发送至 <strong>' +
      esc(CFG.mailto) + '</strong>。', 'err')
  }

  function submit() {
    var d = collect()

    // 蜜罐命中：假装成功但不发送
    if (val('hp')) {
      showOk(d)
      return
    }

    if (!validate(d)) {
      setStatus('请检查标红的必填项。', 'err')
      return
    }

    if (!CFG.webhook) {
      demoMode(d)
      return
    }

    submitBtn.disabled = true
    setStatus('正在提交…', 'info')

    function restore() { submitBtn.disabled = false }
    var timer = setTimeout(restore, 12000)

    try {
      fetch(CFG.webhook, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(buildCard(d)),
      })
        .then(function (r) { return r.json() })
        .then(function (j) {
          clearTimeout(timer)
          restore()
          if (j && (j.code === 0 || j.StatusCode === 0 || j.status === 0)) showOk(d)
          else showErr('接收端返回异常：' + ((j && (j.msg || j.StatusMessage)) || '未知'))
        })
        .catch(function () {
          clearTimeout(timer)
          restore()
          showErr('网络或接收端异常')
        })
    } catch (e) {
      clearTimeout(timer)
      restore()
      showErr('提交异常：' + (e && e.message ? e.message : '未知'))
    }
  }

  function copyText() {
    var d = collect()
    var t = plainText(d)
    function fallback() {
      var ta = document.createElement('textarea')
      ta.value = t
      ta.style.position = 'fixed'
      ta.style.left = '-9999px'
      document.body.appendChild(ta)
      ta.select()
      try { document.execCommand('copy') } catch (e) { /* ignore */ }
      document.body.removeChild(ta)
      setStatus('留言内容已复制，请粘贴发送至 <strong>' + esc(CFG.mailto) + '</strong>。', 'info')
    }
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(t).then(function () {
        setStatus('留言内容已复制，请粘贴发送至 <strong>' + esc(CFG.mailto) + '</strong>。', 'info')
      }).catch(fallback)
    } else {
      fallback()
    }
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault()
    submit()
  })
  copyBtn.addEventListener('click', copyText)

  window.__DHJ__ = { submit: submit, copy: copyText, collect: collect, validate: validate }
})
</script>
